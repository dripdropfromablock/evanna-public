cmake_minimum_required(VERSION 3.22.1)
project(evanna_decoder LANGUAGES C CXX)

get_filename_component(EVANNA_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${EVANNA_SOURCE_ROOT}=."
        "-ffile-prefix-map=${CMAKE_SYSROOT}=/android-sysroot")
set(CMAKE_CXX_STANDARD 17)
set(FFMPEG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/ffmpeg/android/${ANDROID_ABI}")
foreach(component avformat avcodec avutil swresample)
    add_library(${component} SHARED IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${FFMPEG}/lib/lib${component}.so")
endforeach()
add_library(evanna_decoder SHARED decoder_jni.cpp)
set(EBUR128 "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/libebur128")
add_library(ebur128 STATIC "${EBUR128}/ebur128.c")
set_target_properties(ebur128 PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 11)
target_include_directories(ebur128 PRIVATE "${EBUR128}/queue")
target_compile_options(ebur128 PRIVATE -Wall -Wextra -fvisibility=hidden)
target_include_directories(evanna_decoder PRIVATE "${FFMPEG}/include" "${EBUR128}")
target_compile_options(evanna_decoder PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(evanna_decoder PRIVATE -Wl,-z,max-page-size=16384 -Wl,--build-id=sha1)
target_link_libraries(evanna_decoder avformat avcodec avutil swresample ebur128 m)
